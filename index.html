구글 스프레드시트 연동

스프레드시트 > 확장 프로그램 > Apps Script 웹 앱 배포 URL을 입력하세요.
구글 시트 URL을 등록하면 학생 명단과 역할을 실시간으로 가져오고 배정 결과를 시트에 안전하게 보관할 수 있습니다.

구글 스프레드시트의 확장 프로그램 > Apps Script에 아래 코드를 붙여넣고 [웹 앱으로 배포] (액세스: 모든 사용자) 하세요. 

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  // A열: 학생이름, B열: 역할이름, C열: 아이콘, D열: 설명
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(String(data[i][0]).trim());
    if (data[i][1]) {
      roles.push({
        id: i,
        name: String(data[i][1]).trim(),
        icon: data[i][2] ? String(data[i][2]).trim() : '⭐',
        desc: data[i][3] ? String(data[i][3]).trim() : ''
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('배정결과')
              || SpreadsheetApp.getActiveSpreadsheet().insertSheet('배정결과');
  sheet.clear();
  sheet.appendRow(['번호', '학생 이름', '역할', '아이콘', '설명', '배정일시']);
  data.assignments.forEach(function(item) {
    var role = item.role || {};
    sheet.appendRow([
      item.number || 1,
      item.studentName || '',
      item.roleName || role.name || '',
      role.icon || '⭐',
      role.desc || '',
      new Date()
    ]);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
         .setMimeType(ContentService.MimeType.JSON);
}
